  <script>
    function createTable() {
      const tbody = document.getElementById('tableBody');
      tbody.innerHTML = '';
      for (let i = 1; i <= 14; i++) {
        const tr = document.createElement('tr');
        tr.innerHTML = `
          <td class="row-num">${i}</td>
          <td class="col-a"><input type="number" step="0.01" class="price_input" value="0" onfocus="this.select()" oninput="calculateRow(this)"></td>
          <td class="col-z"><input type="number" step="0.01" class="disc_perc_input" value="0" onfocus="this.select()" oninput="calculateRow(this)"></td>
          <td class="col-x row-val disc_price_cell">0.00</td>
          <td class="col-gst">
            <select class="gst_mode_select" onchange="calculateRow(this)">
              <option value="18">+18% GST</option>
              <option value="0">GST INCL.</option>
            </select>
            <div class="row-val gst_cell" style="font-size: 11px; font-weight: bold; margin-top: 2px;">0.00</div>
          </td>
          <td class="col-b"><input type="number" step="0.01" class="margin_input" value="0" onfocus="this.select()" oninput="calculateRow(this)"></td>
          <td class="col-qty"><input type="number" step="1" min="1" class="qty_input" value="1" onfocus="this.select()" oninput="calculateRow(this)"></td>
          <td class="col-c row-val total_cell">0.00</td>
          <td class="col-del"><button class="del-btn" onclick="clearRow(this)">×</button></td>
        `;
        tbody.appendChild(tr);
      }
    }

    function calculateRow(inputElement) {
      const row = inputElement.closest('tr');
      const price = parseFloat(row.querySelector('.price_input').value) || 0;
      const discPerc = parseFloat(row.querySelector('.disc_perc_input').value) || 0;
      const gstMode = parseFloat(row.querySelector('.gst_mode_select').value) || 0;
      const margin = parseFloat(row.querySelector('.margin_input').value) || 0;
      const qty = parseFloat(row.querySelector('.qty_input').value) || 1;

      // 1. Discount Price: X = A - (A * (Z/100))
      const discPrice = price - (price * (discPerc / 100));
      row.querySelector('.disc_price_cell').textContent = discPrice.toFixed(2);

      // 2. Price + GST or GST Included
      const gstPrice = gstMode === 18 ? discPrice + (discPrice * 0.18) : discPrice;
      row.querySelector('.gst_cell').textContent = gstPrice.toFixed(2);

      // 3. Final Total: C = (Unit Price after GST + Margin) * Quantity
      const finalTotal = (gstPrice + margin) * qty;
      row.querySelector('.total_cell').textContent = finalTotal.toFixed(2);

      calculateTotals();
    }

    function calculateTotals() {
      let totalPrice = 0;
      let totalDiscPerc = 0;
      let totalDiscPrice = 0;
      let totalGstPrice = 0;
      let totalMargin = 0;
      let totalQty = 0;
      let totalFinal = 0;

      document.querySelectorAll('#tableBody tr').forEach(row => {
        const price = parseFloat(row.querySelector('.price_input').value) || 0;
        const discPerc = parseFloat(row.querySelector('.disc_perc_input').value) || 0;
        const discPrice = parseFloat(row.querySelector('.disc_price_cell').textContent) || 0;
        const gstPrice = parseFloat(row.querySelector('.gst_cell').textContent) || 0;
        const margin = parseFloat(row.querySelector('.margin_input').value) || 0;
        const qty = parseFloat(row.querySelector('.qty_input').value) || 0;
        const finalTotal = parseFloat(row.querySelector('.total_cell').textContent) || 0;

        totalPrice += price;
        totalDiscPerc += discPerc;
        totalDiscPrice += discPrice;
        totalGstPrice += gstPrice;
        totalMargin += margin;
        if(price > 0 || margin > 0) totalQty += qty;
        totalFinal += finalTotal;
      });

      document.getElementById('total-price').textContent = totalPrice.toLocaleString('en-IN', {minimumFractionDigits: 0, maximumFractionDigits: 0});
      document.getElementById('total-disc_perc').textContent = totalDiscPerc.toLocaleString('en-IN', {minimumFractionDigits: 2, maximumFractionDigits: 2});
      document.getElementById('total-disc_price').textContent = totalDiscPrice.toLocaleString('en-IN', {minimumFractionDigits: 2, maximumFractionDigits: 2});
      document.getElementById('total-gst').textContent = totalGstPrice.toLocaleString('en-IN', {minimumFractionDigits: 2, maximumFractionDigits: 2});
      document.getElementById('total-margin').textContent = totalMargin.toLocaleString('en-IN', {minimumFractionDigits: 0, maximumFractionDigits: 0});
      document.getElementById('total-qty').textContent = totalQty;
      document.getElementById('total-total').textContent = totalFinal.toLocaleString('en-IN', {minimumFractionDigits: 2, maximumFractionDigits: 2});
    }

    function clearRow(btnElement) {
      const row = btnElement.closest('tr');
      row.querySelector('.price_input').value = 0;
      row.querySelector('.disc_perc_input').value = 0;
      row.querySelector('.gst_mode_select').value = "18";
      row.querySelector('.margin_input').value = 0;
      row.querySelector('.qty_input').value = 1;
      calculateRow(row.querySelector('.price_input'));
    }

    function resetCalculator() {
      createTable();
      calculateTotals();
    }

    function copyTotals() {
      const totVal = document.getElementById('total-total').textContent;
      navigator.clipboard.writeText(totVal);
      alert('Total copied: ?' + totVal);
    }

    createTable();
